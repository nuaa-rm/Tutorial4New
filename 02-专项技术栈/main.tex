\chapter*{编者说明}
本卷是视觉组主技能书。组内三个方向为\textbf{自瞄}、\textbf{雷达}、\textbf{导航}；
OpenCV、深度学习、CUDA 等为共用技能，按方向组合选修，不要求一次学完。
\textbf{前置条件}：通过卷一通识考核，具备 Linux、现代 C++ 与 CMake 基础。
标 $*$ 的小节为选修。

\noindent\textbf{推荐选课（示意）：}
\begin{itemize}
    \item \textbf{自瞄}：ROS2 + OpenCV + 深度学习 + 相机 + 滤波 + BA + MPC（CUDA 建议）
    \item \textbf{雷达}：ROS2 + 多机通信 + SDR/电磁波解析 + OpenCV + 深度学习 + 相机（CUDA / 滤波建议）
    \item \textbf{导航}：ROS2 + 导航章（相机 / 滤波 / 多机通信建议）；赛场建图见卷三
\end{itemize}

\chapter{多机通信方法}

\begin{introduction}
  \item 学习目标：在 ROS2 之外掌握本机 TCP 与串口帧通信；理解粘包拆包、CRC 重同步、latest-value 与连接代次；能设计简单的故障隔离。
  \item 建议学时：约 2--3 天。
  \item 适用方向：雷达站多进程、反制双机、与裁判/电控串口对接；自瞄/导航中的跨板通信亦可参考。
\end{introduction}

\input{chapters/J_comm.tex}

\section{本章验收}
\begin{enumerate}
    \item 说明 TCP 字节流为何必须自行组帧，并举粘包与半包各一例。
    \item 对比无限 FIFO 与 latest-value 队列，说明雷达坐标类数据更适合哪一种。
    \item 设计「裁判串口与云台串口分离」的理由（一句话即可）。
\end{enumerate}

\chapter{SDR 与电磁波解析}

\begin{introduction}
  \item 学习目标：理解 SDR / IQ 基本概念；说明分阶段射频状态机；能陈述从 IQ 到 CRC 有效帧的解析步骤；知晓射频合规边界。
  \item 建议学时：约 3--5 天。
  \item 本卷边界：不要求实现完整赛场解调器；具体频点、波形与帧定义以规则及队内仓库为准。与导航章「激光雷达」无关。
\end{introduction}

\input{chapters/K_sdr.tex}

\section{本章验收}
\begin{enumerate}
    \item 解释 I、Q 两路样本的含义，以及采样率与可观察带宽的关系（定性即可）。
    \item 画出「一级干扰 $\rightarrow$ 二级干扰 $\rightarrow$ 信息波」单通道切换的状态顺序，并说明为何不同时解两路。
    \item 列举解析流水线中同步、解调、CRC 三步各自的作用。
\end{enumerate}

\chapter{OpenCV 与图像处理基础}

\begin{introduction}
  \item 学习目标：掌握 \texttt{cv::Mat} 与读写；完成 HSV 分割、阈值、滤波、形态学与轮廓筛选；能用摄像头取流，并用 \texttt{cv\_bridge} 接入 ROS2。
  \item 建议学时：约 1--1.5 周。
  \item 本卷边界：装甲板识别案例与赛场调参见卷三；本卷聚焦 OpenCV 库能力与 C++ 接口。
\end{introduction}

\input{chapters/B_opencv.tex}

\section{本章验收}
\begin{enumerate}
    \item 用 CMake 链接 OpenCV，读取本地图像或摄像头并显示。
    \item 在 HSV 空间完成颜色分割（若为红色，正确处理色环两端），再配合形态学去噪。
    \item 对二值图或 Canny 结果做轮廓提取，用面积（及可选长宽比）筛选后绘制 \texttt{minAreaRect}。
    \item 说明 BGR 与 HSV 的差异，以及浅拷贝与 \texttt{clone} 的区别。
\end{enumerate}

\chapter{CUDA 与 GPU 加速}
