cmake_minimum_required(VERSION 3.18)
project(vector_lab LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(CUDAToolkit REQUIRED)
add_executable(cpu_demo cpu_demo.cpp)
foreach(name device_info vec_add shared_sum error_demo pipeline)
    add_executable(${name} ${name}.cu)
    target_link_libraries(${name} PRIVATE CUDA::cudart)
    target_compile_options(${name} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endforeach()
