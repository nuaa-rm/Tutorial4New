cmake_minimum_required(VERSION 3.16)
project(image_lab_demo LANGUAGES CXX)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui videoio features2d)
message(STATUS "OpenCV: ${OpenCV_VERSION}")
foreach(name version mat viewer video io hsv threshold filter morphology contour resize features sample pipeline)
    add_executable(${name}_demo src/${name}_demo.cpp)
    target_compile_features(${name}_demo PRIVATE cxx_std_17)
    target_include_directories(${name}_demo PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(${name}_demo PRIVATE ${OpenCV_LIBS})
endforeach()
